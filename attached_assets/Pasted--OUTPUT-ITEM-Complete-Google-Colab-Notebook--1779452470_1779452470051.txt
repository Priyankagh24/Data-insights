"""
## Section 3: Initial Data Audit

Produces a quick statistical overview of each column before any cleaning:
- Null / empty counts
- Unique value counts
- Sample values for inspection
"""

def initial_audit(df: pd.DataFrame) -> pd.DataFrame:
    """
    Generate a column-level audit summary for a raw DataFrame.
    All columns are expected to be string dtype at this stage.
    """
    rows = []
    for col in df.columns:
        null_count = df[col].isna().sum()
        empty_count = (df[col].astype(str).str.strip() == "").sum()
        unique_count = df[col].nunique()
        top_values = df[col].value_counts().head(3).to_dict()
        rows.append({
            "column": col,
            "total": len(df),
            "nulls": null_count,
            "empty_strings": empty_count,
            "missing_total": null_count + empty_count,
            "unique_values": unique_count,
            "top_3_values": str(top_values),
        })
    return pd.DataFrame(rows)

def get_txn_id(row: pd.Series) -> str:

# ====================================================
# SECTION 5 — Vendor Canonicalization
# ====================================================
"""
## Section 5: Vendor Canonicalization

Maps all vendor_raw values to canonical names.
For blank vendors, attempts inference from the description field.

Approach:
1. Exact lookup: lowercase(vendor_raw) against VENDOR_MAP
2. Fuzzy fallback: check if any known alias is a substring of the raw value
3. Blank vendor: scan description for known aliases
4. Still unresolved: log CRITICAL; exclude from load
"""

def infer_vendor_from_description(description: str) -> str | None:
    """
    Attempt to infer canonical vendor from an expense description.
    Searches for known alias substrings (case-insensitive).
    Returns canonical name if found, else None.
    """
    desc_lower = description.lower().strip()
    for alias, canonical in VENDOR_MAP.items():
        if alias in desc_lower:
            return canonical
    return None


def normalize_vendor(vendor_raw: str, description: str, txn_id: str) -> str:
    """
    Resolve vendor_raw to its canonical vendor name.

    Parameters
    ----------
    vendor_raw  : Raw vendor string from source CSV
    description : Expense description (used for inference when vendor is blank)
    txn_id      : Transaction ID (for logging)

    Returns
    -------
    Canonical vendor name, or 'UNKNOWN' if unresolvable.
    """
    s = str(vendor_raw).strip()

    # ── Blank vendor ──────────────────────────────
    if s == "" or s.lower() in ("none", "nan", "n/a"):
        inferred = infer_vendor_from_description(str(description))
        if inferred:
            log_issue(
                txn_id, "vendor_raw", "BLANK_VENDOR_INFERRED", "WARNING",
                s,
                f"WARNING — vendor blank; inferred '{inferred}' from description",
            )
            return inferred
        else:
            log_issue(
                txn_id, "vendor_raw", "BLANK_VENDOR_UNRESOLVED", "CRITICAL",
                s,
                "EXCLUDED — vendor blank; cannot be inferred from description",
            )
            return "UNRESOLVED"

    # ── Exact lookup ──────────────────────────────
    key = s.lower().strip()
    if key in VENDOR_MAP:
        canonical = VENDOR_MAP[key]
        if canonical != s:
            log_issue(
                txn_id, "vendor_raw", "VENDOR_ALIAS_RESOLVED", "INFO",
                s,
                f"NORMALISED — '{s}' → '{canonical}'",
            )
        return canonical

    # ── Fuzzy fallback ────────────────────────────
    for alias, canonical in VENDOR_MAP.items():
        if alias in key:
            log_issue(
                txn_id, "vendor_raw", "VENDOR_ALIAS_RESOLVED", "INFO",
                s,
                f"NORMALISED (fuzzy) — '{s}' → '{canonical}'",
            )
            return canonical

    # ── Unknown vendor ────────────────────────────
    log_issue(
        txn_id, "vendor_raw", "UNRECOGNISED_VENDOR", "WARNING",
        s,
        f"WARNING — '{s}' not in canonical list; loaded as-is",
    )
    return s

# ====================================================
# SECTION 6 — Date Parsing
# ====================================================
"""
## Section 6: Date Parsing

parse_date() handles every date format encountered in enterprise exports:
- ISO 8601: YYYY-MM-DD, YYYY/MM/DD
- European: DD-MM-YYYY, DD/MM/YYYY
- US: MM/DD/YYYY
- Textual: "12 Apr 2026", "April 12, 2026", "12-Apr-2026"
- Unix epoch (10-digit seconds or 13-digit milliseconds)
- Mixed separators, trailing timestamps, whitespace
- Null / empty / sentinel values

Output: YYYY-MM-DD string or None (with issue logged).

Ambiguity assumption: when a date could be either MM/DD/YYYY or DD/MM/YYYY
and the first part is ≤ 12, we default to DD/MM/YYYY (European convention)
since the client is an Indian company.
"""

def parse_date(raw: str, txn_id: str, field: str) -> str | None:
    """
    Coerce raw date string to YYYY-MM-DD.

    Returns
    -------
    str | None
        Normalised date string, or None if unparseable.
    """
    if raw is None:
        return None

    s = str(raw).strip()
    if s in ("", "None", "NaT", "nan", "NaN", "null", "N/A"):
        return None

    original = s

    # ── 1. Unix epoch (10-digit seconds) ──────────
    if re.fullmatch(r"\d{9,10}", s):
        try:
            return datetime.utcfromtimestamp(int(s)).strftime("%Y-%m-%d")
        except Exception:
            pass

    # ── 2. Millisecond epoch (13-digit) ───────────
    if re.fullmatch(r"\d{13}", s):
        try:
            return datetime.utcfromtimestamp(int(s) / 1000).strftime("%Y-%m-%d")
        except Exception:
            pass

    # ── 3. ISO YYYY-MM-DD or YYYY/MM/DD ───────────
    iso_match = re.match(r"(\d{4})[-/](\d{1,2})[-/](\d{1,2})", s)
    if iso_match:
        y, m, d = iso_match.groups()
        try:
            return datetime(int(y), int(m), int(d)).strftime("%Y-%m-%d")
        except ValueError:
            pass

    # ── 4. Normalise separators for further parsing ─
    s_norm = re.sub(r"[./\s]+", "-", s).split("T")[0].split(" ")[0]

    # ── 5. DD-MM-YYYY ─────────────────────────────
    if re.fullmatch(r"\d{1,2}-\d{1,2}-\d{4}", s_norm):
        for fmt in ("%d-%m-%Y", "%m-%d-%Y"):
            try:
                return datetime.strptime(s_norm, fmt).strftime("%Y-%m-%d")
            except ValueError:
                pass

    # ── 6. Textual month formats ──────────────────
    for fmt in (
        "%d %b %Y", "%d-%b-%Y", "%d %B %Y", "%d-%B-%Y",
        "%b %d %Y", "%B %d %Y", "%b %d, %Y", "%B %d, %Y",
        "%d/%b/%Y", "%d/%B/%Y", "%b-%d-%Y", "%B-%d-%Y",
    ):
        try:
            return datetime.strptime(original, fmt).strftime("%Y-%m-%d")
        except ValueError:
            pass

    # ── 7. pandas last-resort ─────────────────────
    try:
        return pd.to_datetime(original, dayfirst=True).strftime("%Y-%m-%d")
    except Exception:
        pass

    return None


def apply_date_parsing(df: pd.DataFrame, col: str, required: bool = True) -> pd.Series:
    """
    Apply parse_date to an entire column, logging issues per row.

    Parameters
    ----------
    col      : Column name to parse
    required : If True, unparseable values are CRITICAL; else INFO
    """
    results = []
    for idx, row in df.iterrows():
        txn_id = get_txn_id(row)
        raw = str(row[col]).strip() if col in df.columns else ""
        parsed = parse_date(raw, txn_id, col)

        if parsed is None:
            sev = "CRITICAL" if required else "INFO"
            action = (
                "EXCLUDED — entry_date is mandatory"
                if required
                else "LOADED — submission_date left NULL"
            )
            log_issue(txn_id, col, "UNPARSEABLE_DATE", sev, raw, action)
        elif parsed != raw:
            log_issue(
                txn_id, col, "DATE_FORMAT_NORMALISED", "INFO",
                raw, f"NORMALISED — '{raw}' → '{parsed}'"
            )
        results.append(parsed)
    return pd.Series(results, index=df.index)


print("✅ Section 6 complete — Date parsing functions defined.")


# ====================================================
# SECTION 7 — Amount Parsing
# ====================================================
"""
## Section 7: Amount Parsing

parse_amount() extracts a clean numeric float plus the originating currency
from any encoding found in enterprise CSV exports.

Handles all specified formats:
  ₹1,50,000 | $1,200.50 | EUR 49.99 | AED 32,000 | SGD 7800
  (5,200) → refund | -250 → negative | 22,00,000 → Indian notation
  TBD | N/A | blank → CRITICAL exclusion
  Text contamination (stripped)

Returns: (amount_float | None, currency_code | 'UNKNOWN')
"""

def parse_amount(raw: str, txn_id: str) -> tuple:
    """
    Extract (numeric_amount, currency_code) from a raw amount string.

    Returns (None, 'UNKNOWN') for unresolvable values.
    The caller is responsible for INR conversion.
    """
    if raw is None:
        return None, "UNKNOWN"

    s = str(raw).strip()

    # ── Unresolvable sentinels ─────────────────────
    if s.upper() in ("TBD", "N/A", "NA", "", "NONE", "NULL", "NAN"):
        log_issue(
            txn_id, "amount_raw", "NON_NUMERIC_VALUE", "CRITICAL",
            raw,
            "EXCLUDED — amount unresolvable without source invoice",
        )
        return None, "UNKNOWN"

    # ── Detect currency from symbol ────────────────
    currency = "INR"   # default assumption for Indian company
    for sym, code in SYMBOL_TO_CODE.items():
        if sym in s:
            currency = code
            break

    # ── Detect currency from embedded code ─────────
    code_match = CURRENCY_CODE_RE.search(s)
    if code_match:
        currency = code_match.group(1).upper()

    # ── Strip symbols and currency codes ──────────
    cleaned = s
    for sym in SYMBOL_TO_CODE:
        cleaned = cleaned.replace(sym, "")
    cleaned = CURRENCY_CODE_RE.sub("", cleaned).strip()

    # ── Handle parentheses negatives: (5,200) ──────
    is_negative = False
    paren_match = re.fullmatch(r"\(\s*([0-9,.\s]+)\s*\)", cleaned.strip())
    if paren_match:
        cleaned = paren_match.group(1)
        is_negative = True
    elif cleaned.startswith("-"):
        is_negative = True
        cleaned = cleaned.lstrip("-").strip()

    # ── Remove all commas (Indian + Western notation)
    cleaned = re.sub(r",", "", cleaned).strip()

    # ── Extract numeric value ──────────────────────
    num_match = re.search(r"\d+\.?\d*", cleaned)
    if not num_match:
        log_issue(
            txn_id, "amount_raw", "NON_NUMERIC_VALUE", "CRITICAL",
            raw,
            "EXCLUDED — no extractable numeric value after cleaning",
        )
        return None, "UNKNOWN"

    amount = float(num_match.group())
    if is_negative:
        amount = -amount

    # ── Validate currency in rate table ───────────
    if currency not in EXCHANGE_RATES:
        log_issue(
            txn_id, "amount_raw", "UNKNOWN_CURRENCY", "WARNING",
            raw,
            f"WARNING — currency '{currency}' not in rate table; defaulted to INR",
        )
        currency = "INR"

    # ── Log normalisation if value changed ─────────
    if str(raw).strip() not in (str(amount), f"{amount:.2f}"):
        log_issue(
            txn_id, "amount_raw", "AMOUNT_NORMALISED", "INFO",
            raw,
            f"NORMALISED — '{raw}' → {amount} {currency}",
        )

    return amount, currency


def convert_to_inr(amount: float, currency: str) -> tuple:
    """
    Convert amount to INR using static reference rates.

    Returns (amount_inr, exchange_rate_used).
    exchange_rate_used is None for INR→INR conversions.
    """
    if amount is None:
        return None, None
    rate = EXCHANGE_RATES.get(currency, 1.0)
    inr = round(amount * rate, 2)
    return inr, (rate if currency != "INR" else None)


print("✅ Section 7 complete — Amount parsing functions defined.")

"""
## Section 8: Currency Normalization

Applies amount parsing and INR conversion to the full dataset.
Produces three new columns:
  - amount_numeric     : clean float in original currency
  - original_currency  : detected currency code (INR/USD/EUR/GBP/SGD/AED)
  - exchange_rate_used : rate applied (NULL for INR→INR)
  - amount_inr         : final INR amount
"""

def apply_currency_normalization(df: pd.DataFrame) -> pd.DataFrame:
    """Apply parse_amount and INR conversion to every row."""
    amounts, currencies, rates, inr_amounts = [], [], [], []

    for idx, row in df.iterrows():
        txn_id = get_txn_id(row)
        raw = str(row.get("amount_raw", "")).strip()

        amount, currency = parse_amount(raw, txn_id)
        inr, rate = convert_to_inr(amount, currency)

        amounts.append(amount)
        currencies.append(currency)
        rates.append(rate)
        inr_amounts.append(inr)

    df = df.copy()
    df["amount_numeric"]    = amounts
    df["original_currency"] = currencies
    df["exchange_rate_used"] = rates
    df["amount_inr"]        = inr_amounts
    return df


print("✅ Section 8 complete — Currency normalization functions defined.")

# SECTION 9 — Duplicate Detection
# ====================================================
"""
## Section 9: Duplicate Detection

Methodology:
─────────────────────────────────────────────────────
EXACT DUPLICATE: All five fields identical
  vendor_canonical, amount_numeric, employee_name, txn_date, description

  Action: Mark all but the earliest txn_ref as is_duplicate=True.
  Set duplicate_of = txn_ref of the retained record.

NEAR DUPLICATE: Four fields identical (description excluded)
  vendor_canonical, amount_numeric, employee_name, txn_date

  When description differs, intent is ambiguous (could be a resubmission
  with a corrected note, or a genuine separate expense on the same day).
  Action: Flag ALL records in the group as under_review=True, is_flagged=True.

Rule is fully generalised — no row IDs hardcoded.
Decision: earliest txn_ref (lexicographic) is the "original" retained record.
"""

def detect_duplicates(df: pd.DataFrame) -> pd.DataFrame:
    """
    Identify exact and near-exact duplicate transactions.
    Populates: is_duplicate, duplicate_of, under_review, is_flagged, flag_reason.
    """
    df = df.copy()
    df["is_duplicate"]  = False
    df["duplicate_of"]  = None
    df["under_review"]  = False
    df["is_flagged"]    = False
    df["flag_reason"]   = None

    EXACT_COLS = ["vendor_canonical", "amount_numeric", "employee_name", "txn_date", "description"]
    NEAR_COLS  = ["vendor_canonical", "amount_numeric", "employee_name", "txn_date"]

    exact_cols = [c for c in EXACT_COLS if c in df.columns]
    near_cols  = [c for c in NEAR_COLS  if c in df.columns]

    if not near_cols:
        print("⚠️  Insufficient columns for duplicate detection.")
        return df

    id_col = "txn_id"

    # ── EXACT DUPLICATES ──────────────────────────
    if exact_cols:
        exact_grp = df.groupby(exact_cols)[id_col].apply(list)
        for key_vals, ids in exact_grp.items():
            if len(ids) > 1:
                original = sorted(ids)[0]      # earliest txn_ref alphabetically
                for dup in sorted(ids)[1:]:
                    mask = df[id_col] == dup
                    df.loc[mask, "is_duplicate"] = True
                    df.loc[mask, "duplicate_of"] = original
                    df.loc[mask, "is_flagged"]   = True
                    existing_reason = df.loc[mask, "flag_reason"].values[0]
                    new_reason = f"EXACT_DUPLICATE of {original}"
                    df.loc[mask, "flag_reason"] = (
                        f"{existing_reason}; {new_reason}"
                        if existing_reason else new_reason
                    )
                    log_issue(
                        dup, id_col, "EXACT_DUPLICATE", "WARNING",
                        str(dup),
                        f"FLAGGED — exact duplicate of {original}; is_duplicate=True",
                    )

    # ── NEAR DUPLICATES ───────────────────────────
    near_grp = df.groupby(near_cols)[id_col].apply(list)
    for key_vals, ids in near_grp.items():
        if len(ids) > 1:
            # Only flag those not already marked as exact duplicates
            non_exact = [i for i in ids if not df.loc[df[id_col] == i, "is_duplicate"].values[0]]
            if len(non_exact) > 1:
                for tid in non_exact:
                    mask = df[id_col] == tid
                    df.loc[mask, "under_review"] = True
                    df.loc[mask, "is_flagged"]   = True
                    existing_reason = df.loc[mask, "flag_reason"].values[0]
                    new_reason = "NEAR_DUPLICATE (same vendor/amount/employee/date; description differs)"
                    df.loc[mask, "flag_reason"] = (
                        f"{existing_reason}; {new_reason}"
                        if existing_reason else new_reason
                    )
                    log_issue(
                        tid, id_col, "NEAR_DUPLICATE", "WARNING",
                        str(tid),
                        "FLAGGED — near-duplicate detected; under_review=True",
                    )

    exact_count = df["is_duplicate"].sum()
    near_count  = df["under_review"].sum()
    print(f"  Exact duplicates  : {exact_count}")
    print(f"  Near duplicates   : {near_count}")
    return df


print("✅ Section 9 complete — Duplicate detection functions defined.")

# SECTION 10 — Personal Expense Detection
# ====================================================
"""
## Section 10: Personal Expense Detection

Generalised rule-based detection using keyword regex patterns.
No row IDs are hardcoded. Rules apply identically to all rows.

Flags:
  is_personal  = True when a personal keyword is matched
  is_flagged   = True (combined with flag_reason)

Limitation: keyword matching may produce false positives (e.g., "bar chart"
in a data analytics description). This is logged as WARNING for human review,
not automatically excluded.
"""

def detect_personal_expenses(df: pd.DataFrame) -> pd.DataFrame:
    """Apply personal expense detection to the description column."""
    df = df.copy()
    if "is_personal" not in df.columns:
        df["is_personal"] = False

    for idx, row in df.iterrows():
        txn_id = get_txn_id(row)
        desc = str(row.get("description", ""))
        match = PERSONAL_PATTERN.search(desc)
        if match:
            df.at[idx, "is_personal"] = True
            df.at[idx, "is_flagged"]  = True
            keyword = match.group(0)
            existing = df.at[idx, "flag_reason"]
            new_reason = f"PERSONAL_EXPENSE (keyword: '{keyword}')"
            df.at[idx, "flag_reason"] = (
                f"{existing}; {new_reason}" if existing else new_reason
            )
            log_issue(
                txn_id, "description", "PERSONAL_EXPENSE_DETECTED", "WARNING",
                desc[:100],
                f"FLAGGED — keyword match: '{keyword}'; is_personal=True",
            )

    personal_count = df["is_personal"].sum()
    print(f"  Personal expenses detected: {personal_count}")
    return df


print("✅ Section 10 complete — Personal expense detection defined.")

# SECTION 11 — Department Standardization
# ====================================================
"""
## Section 11: Department Standardization

Maps all department variants to five canonical names:
  Engineering | Sales | Product | Operations | Finance

Handles: abbreviations (eng, fin, ops), typos (engg), casing, punctuation.
Logs WARNING for values that cannot be mapped.
"""

def standardize_department(dept_raw: str, txn_id: str) -> str:
    """
    Map department_raw to canonical department name.

    Returns canonical name, or the original value if not mappable.
    """
    s = str(dept_raw).strip()
    if s in ("", "None", "nan"):
        log_issue(
            txn_id, "department", "BLANK_DEPARTMENT", "WARNING",
            s,
            "WARNING — department blank; requires manual assignment",
        )
        return ""

    key = s.lower().rstrip(".")
    canonical = DEPT_MAP.get(key)

    if canonical is None:
        log_issue(
            txn_id, "department", "UNRECOGNISED_DEPARTMENT", "WARNING",
            s,
            f"WARNING — '{s}' not mappable; loaded as-is",
        )
        return s
    elif canonical != s:
        log_issue(
            txn_id, "department", "DEPARTMENT_NORMALISED", "INFO",
            s,
            f"NORMALISED — '{s}' → '{canonical}'",
        )
    return canonical


def apply_department_standardization(df: pd.DataFrame) -> pd.DataFrame:
    """Apply standardize_department to the entire department column."""
    df = df.copy()
    df["department"] = df.apply(
        lambda row: standardize_department(
            str(row.get("department", "")), get_txn_id(row)
        ),
        axis=1,
    )
    return df


print("✅ Section 11 complete — Department standardization defined.")

# SECTION 12 — Validation Checks
# ====================================================
"""
## Section 12: Validation Checks

Final integrity checks after all transformations:
  - Blank required fields (txn_id, txn_date, vendor, amount)
  - Completely blank rows
  - amount_inr = 0 (suspicious)
  - Negative amounts not marked as refunds
"""

def run_validation_checks(df: pd.DataFrame, excluded_ids: set) -> set:
    """
    Run final integrity validations.
    Updates excluded_ids in-place and appends to quality_issues.

    Returns
    -------
    set: Updated set of excluded txn_ids.
    """
    REQUIRED = ["txn_id", "txn_date", "vendor_raw", "amount_raw"]

    for idx, row in df.iterrows():
        txn_id = get_txn_id(row)

        # ── Completely blank rows ──────────────────
        key_fields = ["txn_id", "vendor_raw", "amount_raw", "txn_date"]
        all_blank = all(
            str(row.get(f, "")).strip() in ("", "None", "nan")
            for f in key_fields if f in df.columns
        )
        if all_blank:
            log_issue(
                txn_id, "ALL", "BLANK_ROW", "CRITICAL",
                "",
                "EXCLUDED — entirely blank pipeline row",
            )
            excluded_ids.add(txn_id)
            continue

        # ── Blank required fields ──────────────────
        for field in REQUIRED:
            if field not in df.columns:
                continue
            val = str(row.get(field, "")).strip()
            if val in ("", "None", "nan"):
                sev = "CRITICAL" if field in ("txn_id", "txn_date", "amount_raw") else "WARNING"
                if sev == "CRITICAL":
                    excluded_ids.add(txn_id)
                log_issue(
                    txn_id, field, "BLANK_REQUIRED_FIELD", sev,
                    "",
                    f"{'EXCLUDED' if sev == 'CRITICAL' else 'WARNING'} — '{field}' is blank",
                )

        # ── Zero amount ────────────────────────────
        if "amount_inr" in df.columns:
            inr = row.get("amount_inr")
            if inr == 0:
                log_issue(
                    txn_id, "amount_inr", "ZERO_AMOUNT", "WARNING",
                    "0",
                    "WARNING — amount_inr = 0; possible data entry error",
                )

    return excluded_ids

    #       max_tokens=200,
    #       system=EXPENSE_CLASSIFICATION_SYSTEM_PROMPT,
    #       messages=[{"role": "user", "content": description}]
    #   )
    #   return json.loads(response.content[0].text)

    desc_lower = description.lower()

    # Cloud Infrastructure
    cloud_kw = ["ec2", "s3", "aws", "azure", "gcp", "cloud", "rds", "lambda",
                "kubernetes", "docker", "server", "storage bucket", "cdn", "vpc"]
    # SaaS
    saas_kw  = ["zoom", "notion", "salesforce", "adobe", "slack", "github",
                "jira", "confluence", "figma", "subscription", "license", "renewal",
                "creative cloud", "workspace", "hubspot"]
    # Travel
    travel_kw = ["flight", "cab", "uber", "ola", "taxi", "hotel", "indigo",
                 "airline", "train", "bus", "transport", "airport", "travel",
                 "boarding", "rapido"]
    # Food
    food_kw  = ["swiggy", "zomato", "meal", "lunch", "dinner", "catering",
                "restaurant", "food", "breakfast", "coffee", "tea"]
    # Hardware
    hw_kw    = ["laptop", "monitor", "keyboard", "mouse", "webcam", "headset",
                "printer", "charger", "desk", "chair", "equipment", "device",
                "computer", "flipkart", "hardware", "peripheral"]
    # Finance
    fin_kw   = ["razorpay", "stripe", "gateway", "payment fee", "bank",
                "wire transfer", "forex", "banking", "transaction fee"]
    # Personal
    pers_kw  = ["personal", "gym", "netflix", "amazon prime", "grocery",
                "hotstar", "spotify", "salon", "spa", "shopping", "movie",
                "cinema", "vacation", "holiday"]

    def score(keywords):
        return sum(1 for kw in keywords if kw in desc_lower)

    scores = {
        "Cloud Infrastructure"  : score(cloud_kw),
        "SaaS Subscriptions"    : score(saas_kw),
        "Travel & Transport"    : score(travel_kw),
        "Meals & Catering"      : score(food_kw),
        "Hardware & Equipment"  : score(hw_kw),
        "Finance & Banking"     : score(fin_kw),
        "Personal Expense"      : score(pers_kw),
    }

    best_cat = max(scores, key=scores.get)
    best_score = scores[best_cat]

    if best_score == 0:
        return {
            "category": "Miscellaneous",
            "confidence": 0.45,
            "reasoning": "No recognisable keywords matched any known expense category.",
        }

    # Scale confidence: more keyword hits → higher confidence
    confidence = min(0.95, 0.60 + best_score * 0.12)

    # Penalise ambiguity: if second-best is close
    sorted_scores = sorted(scores.values(), reverse=True)
    if len(sorted_scores) > 1 and sorted_scores[1] >= best_score - 1 and best_score > 0:
        confidence = max(0.55, confidence - 0.18)

    # Category-specific reasoning
    reasoning_map = {
        "Cloud Infrastructure":  f"Description references cloud compute/storage terms ({', '.join(kw for kw in cloud_kw if kw in desc_lower)}).",
        "SaaS Subscriptions":    f"Description references a known SaaS product or subscription term ({', '.join(kw for kw in saas_kw if kw in desc_lower)}).",
        "Travel & Transport":    f"Description references travel or transport ({', '.join(kw for kw in travel_kw if kw in desc_lower)}).",
        "Meals & Catering":      f"Description references food/catering context ({', '.join(kw for kw in food_kw if kw in desc_lower)}).",
        "Hardware & Equipment":  f"Description references physical hardware or equipment ({', '.join(kw for kw in hw_kw if kw in desc_lower)}).",
        "Finance & Banking":     f"Description references financial services or payment processing ({', '.join(kw for kw in fin_kw if kw in desc_lower)}).",
        "Personal Expense":      f"Description contains personal-use indicators ({', '.join(kw for kw in pers_kw if kw in desc_lower)}).",
    }
    reasoning = reasoning_map.get(best_cat, "Matched by keyword heuristic.")

    return {
        "category": best_cat,
        "confidence": round(confidence, 2),
        "reasoning": reasoning,
    }
